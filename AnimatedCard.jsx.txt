import { motion, useScroll, useTransform } from "framer-motion";
import { useRef } from "react";
import { useNavigate } from "react-router-dom";


export default function AnimatedCard({
  i,
  title,
  description,
  technologies,
  url,
  color,
  btnColor,
  progress,
  range,
  targetScale,
}) {
  const container = useRef(null);
  const navigate=useNavigate()


  const { scrollYProgress } = useScroll({
    target: container,
    offset: ["start end", "start start"],
  });


  const imageScale = useTransform(scrollYProgress, [0, 1], [2, 1]);
  const scale = useTransform(progress, range, [1, targetScale]);


  return (
    <div
      ref={container}
      className="h-[90vh] md:h-screen flex items-center justify-center sticky top-0"
    >
      <motion.div
        style={{
          backgroundColor: color,
          scale,
          top: `calc(-5vh + ${i * 25}px)`,
        }}
        className="
          flex flex-col relative
          md:-top-[25%]
          h-[500px] md:h-[450px]
          w-[95%] sm:w-[90%] md:w-[70%]
          rounded-xl
          p-4 sm:p-6 md:p-10
          origin-top
        "
      >
        {/* Title */}
        <h2 className="text-xl sm:text-2xl text-center font-semibold text-white">
          {title}
        </h2>


        <div className="flex flex-col md:flex-row h-full mt-5 gap-6 md:gap-10">
          {/* TEXT SECTION */}
          <div className="w-full md:w-[40%] flex flex-col gap-5">
            {/* Description */}
            <p
              className="
                text-sm sm:text-base font-medium
                leading-relaxed text-gray-200
                transition-all duration-200
                hover:text-gray-100
              "
            >
              {description}
            </p>


            {/* Technologies Badges */}
          <div className="flex flex-wrap gap-2 sm:gap-3">
            {technologies?.map((tech, index) => (
              <div
                key={index}
                className="
                  px-3 py-2 text-[10px] sm:text-xs font-medium
                  rounded-lg bg-white/10 text-gray-100
                  border border-white/20 backdrop-blur-md
                "
              >
                {tech}
              </div>
            ))}
          </div>


            {/* Buttons */}
            <div className="flex flex-wrap gap-3 mt-2">
              {btnColor?.map((btn, indexbtn) => (
                <motion.button
                  key={indexbtn}
                  onClick={() => navigate("/portfolio")}
                  className={`
                    w-28 sm:w-32 h-10 sm:h-12 rounded-xl
                    flex items-center justify-center
                    text-white font-semibold text-sm sm:text-base
                    shadow-lg shadow-white/40
                    bg-gradient-to-br from-[${btn.from}] to-[${btn.to}]
                  `}
                  whileHover={{
                    scale: 1.1,
                    background: `linear-gradient(135deg, ${btn.hoverFrom}, ${btn.hoverTo})`,
                  }}
                  whileTap={{ scale: 0.95 }}
                >
                  Read More
                </motion.button>
              ))}
            </div>
          </div>


          {/* IMAGE SECTION */}
          <div className="relative w-full md:w-[60%] h-52 sm:h-64 md:h-full rounded-lg overflow-hidden">
            <motion.div className="w-full h-full" style={{ scale: imageScale }}>
              <img
                src={url}
                alt={title}
                className="w-full h-full object-cover"
              />
            </motion.div>
          </div>
        </div>
      </motion.div>
    </div>
  );
}
